## 📊 每日学习热力图

```dataviewjs
const cur = dv.current();

if (cur && cur.date) {
    const startOfWeek = window.moment(cur.date.toString());
    const dataMap = new Map();

    const getColor = (score) => {
        if (!score || score < 20) return "#95a5a6"; 
        if (score < 40) return "#3498db";
        if (score < 60) return "#2ecc71";
        if (score < 80) return "#f1c40f"; // 黄色区间
        return "#e74c3c";
    };

    for (let i = 0; i < 7; i++) {
        // 依次推算周一到周日
        const target = window.moment(startOfWeek).add(i, 'days');
        const y = target.format("YYYY");
        const m = target.format("MM");
        const d = target.format("MM-DD");
        
        // 自动构建路径
        const path = `学习安排/${y}/${m}/${d}.md`;
        const page = dv.page(path);
        
        if (page && page["学习得分"] !== undefined) {
            const score = parseFloat(page["学习得分"]) || 0;
            dataMap.set(i, { score: score.toFixed(1), color: getColor(score) });
        } else {
            dataMap.set(i, { score: "0.0", color: "#95a5a6" });
        }
    }

    let boxes = "";
    const weekLabels = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];
    for (let i = 0; i < 7; i++) {
        const data = dataMap.get(i);
        boxes += `<div style="width:32px;height:32px;background-color:${data.color};border-radius:4px;border:1px solid rgba(0,0,0,0.05);" title="${weekLabels[i]} 得分: ${data.score}"></div>`;
    }

    const container = `
<div style="display:flex; flex-direction:column; align-items:flex-end; padding:10px 0; width:fit-content; margin:0 auto;">
    <div style="display:grid; grid-template-columns:repeat(7, 1fr); gap:8px;">
        ${boxes}
    </div>
    <div style="margin-top:12px; display:flex; align-items:center; gap:8px; font-size:10px; opacity:0.4;">
        <span>Less</span>
        <div style="display:flex; gap:3px;">
            <div style="width:8px; height:8px; background:#95a5a6; border-radius:1px;"></div>
            <div style="width:8px; height:8px; background:#3498db; border-radius:1px;"></div>
            <div style="width:8px; height:8px; background:#2ecc71; border-radius:1px;"></div>
            <div style="width:8px; height:8px; background:#f1c40f; border-radius:1px;"></div>
            <div style="width:8px; height:8px; background:#e74c3c; border-radius:1px;"></div>
        </div>
        <span>More</span>
    </div>
</div>`;

    dv.el("div", container);
} else {
    dv.paragraph("⚠️ YAML 中未找到 date 属性");
}
```
