// 1. 数据提取逻辑
const tasks = dv.current().file.tasks;
let stats = { review: 0, learn: 0, errors: 0, minutes: 0 };
const regex = {
    review: /复习\s*(\d+)/,
    learn: /学\s*(\d+)/,
    error: /错题整理:\s*(\d+)/,
    mins: /(\d+)\s*分钟/
};

tasks.forEach(t => {
    const text = t.text;
    const r = text.match(regex.review), l = text.match(regex.learn), 
          e = text.match(regex.error), m = text.match(regex.mins);
    if (r) stats.review += parseInt(r[1]);
    if (l) stats.learn += parseInt(l[1]);
    if (e) stats.errors += parseInt(e[1]);
    if (m) stats.minutes += parseInt(m[1]);
});

const totalScore = (stats.review * 0.1) + (stats.learn * 0.2) + (stats.errors * 3.0) + (stats.minutes * 0.15);

// --- 新增：自动更新元数据代码 ---
const currentFile = app.vault.getAbstractFileByPath(dv.current().file.path);
if (currentFile) {
    app.fileManager.processFrontMatter(currentFile, (fm) => {
        // 只有当分数发生变化时才更新，避免无限刷新
        const roundedScore = parseFloat(totalScore.toFixed(1));
        if (fm["学习得分"] !== roundedScore) {
            fm["学习得分"] = roundedScore;
        }
    });
}
// ------------------------------

// 2. 状态颜色
let statusColor = "";
if (totalScore < 20) statusColor = "#95a5a6";
else if (totalScore < 40) statusColor = "#3498db";

else if (totalScore < 80) statusColor = "#f1c40f";
else statusColor = "#e74c3c";

// 3. 渲染视图

">
    <div style="display: flex; align-items: baseline; gap: 20px;">
        <span style="font-size: 24px; font-weight: 600; opacity: 0.5; color: var(--text-normal);">总分:</span>
        <span style="font-size: 120px; font-weight: 900; line-height: 0.8; color: var(--text-normal); letter-spacing: -4px;">
            ${totalScore.toFixed(1)}
        </span>
    </div>

    <div style="height: 50px;"></div>
