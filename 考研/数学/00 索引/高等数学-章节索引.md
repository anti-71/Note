```dataviewjs
// 1. 获取所有相关笔记
let pages = dv.pages('"考研/数学/01 高等数学"')
    .filter(p => {
        if (!p.章 || !p.节) return false;
        if (p.file.etags.includes("#例题")) return false;
        return true;
    })
    // 【修改点 1】：强制使用数值排序
    .sort(p => p.index, 'asc', (a, b) => Number(a) - Number(b)); 

// 2. 使用对象聚合数据
let dataMap = {};

for (let p of pages) {
    if (!dataMap[p.章]) dataMap[p.章] = {};
    if (!dataMap[p.章][p.节]) {
        dataMap[p.章][p.节] = {
            index: 9999, 
            sectionLink: p.节, 
            kps: []
        };
    }

    if (p.file.etags.includes("#节")) {
        dataMap[p.章][p.节].sectionLink = dv.fileLink(p.file.path, false, p.节);
        dataMap[p.章][p.节].index = Number(p.index); // 确保存入的是数字
    } else {
        dataMap[p.章][p.节].kps.push(dv.fileLink(p.file.path, false, p.名称 || p.file.name));
    }
}

// 3. 构建表格行
let rows = [];
// 如果章名也是数字，建议也加上数值排序逻辑
let chapterKeys = Object.keys(dataMap).sort((a, b) => {
    let numA = parseInt(a.match(/\d+/)) || 0;
    let numB = parseInt(b.match(/\d+/)) || 0;
    return numA - numB;
}); 

for (let chapterName of chapterKeys) {
    let firstInChapter = true;
    
    let sortedSections = Object.keys(dataMap[chapterName])
        .map(secName => ({ name: secName, ...dataMap[chapterName][secName] }))
        // 【修改点 2】：确保节的 index 比较也是数值类型
        .sort((a, b) => Number(a.index) - Number(b.index));

    for (let sec of sortedSections) {
        rows.push([
            firstInChapter ? `**${chapterName}**` : "",
            `_ ${sec.sectionLink} _`,
            sec.kps.join("<br>") || "（暂无知识点）"
        ]);
        firstInChapter = false;
    }
}

// 4. 渲染
dv.table(["章", "节", "名称"], rows);
```
