# 📐 错题集 · 白板导航

```dataviewjs
// --- Canvas 白板数据 ---
const rootPaths = {
    "高等数学": "考研/数学/05 错题集/01 高等数学",
    "线性代数": "考研/数学/05 错题集/02 线性代数",
    "概率论": "考研/数学/05 错题集/03 概率论与数理统计"
};

const canvasData = {
    "高等数学": [
        "第一章 函数、极限、连续",
        "第二章 一元函数微分学",
        "第三章 一元函数积分学",
        "第四章 多元函数微分学",
        "第五章 二重积分",
        "第六章 微分方程",
        "第七章 无穷级数",
        "第八章 空间解析几何",
        "第九章 多元积分学"
    ],
    "线性代数": [
        "第一章 行列式",
        "第二章 矩阵",
        "第三章 向量",
        "第四章 线性方程组",
        "第五章 特征值和特征向量",
        "第六章 二次型"
    ],
    "概率论": [
        "第一章 随机事件与概率",
        "第二章 一维随机变量及其分布",
        "第三章 多维随机变量及其分布",
        "第四章 随机变量的数字特征",
        "第五章 大数定律与中心极限定理",
        "第六章 数理统计的基本概念",
        "第七章 参数估计与假设检验"
    ]
};

// 科目编号映射
const subjectNum = { "高等数学": "01", "线性代数": "02", "概率论": "03" };

// --- 状态管理 ---
if (!window._errState) {
    window._errState = { subject: "高等数学" };
}
const state = window._errState;

const container = dv.container.createEl("div", { id: "error-dashboard" });

// --- 科目选择 ---
container.createEl("h2", { text: "📚 选择科目", cls: "math-section-title" });
const subRow = container.createEl("div", {
    cls: "sub-row",
    attr: { style: "grid-template-columns: repeat(3, 1fr) !important;" }
});

Object.keys(rootPaths).forEach(sub => {
    const isActive = state.subject === sub;
    const btn = subRow.createEl("div", {
        text: sub,
        cls: `math-btn ${isActive ? "is-active" : ""}`
    });
    btn.onclick = () => { state.subject = sub; render(); };
});

container.createEl("hr", { attr: { style: "margin: 20px 0; opacity: 0.1;" } });

// --- 渲染函数 ---
function render() {
    // 清除旧表格
    const oldTable = container.querySelector("table");
    if (oldTable) oldTable.remove();

    const chapters = canvasData[state.subject];
    if (!chapters || chapters.length === 0) return;

    const num = subjectNum[state.subject];
    const basePath = rootPaths[state.subject];
    const currentPath = dv.current().file.path;

    const table = container.createEl("table", { cls: "math-table" });
    const thead = table.createEl("thead").createEl("tr");
    thead.createEl("th", { text: "章节", attr: { style: "width: 50%;" } });
    thead.createEl("th", { text: "白板文件" });

    const tbody = table.createEl("tbody");

    chapters.forEach(ch => {
        const row = tbody.createEl("tr");
        const tdCh = row.createEl("td", { attr: { style: "text-align: center; font-weight: bold;" } });
        tdCh.setText(ch);

        const tdCanvas = row.createEl("td");
        const canvasContainer = tdCanvas.createEl("div", { attr: { style: "display: flex; flex-wrap: wrap; gap: 8px;" } });

        const canvasPath = `${basePath}/00 白板/${ch}.canvas`;
        const canvasSpan = canvasContainer.createEl("span", { cls: "kp-tag" });
        dv.api.renderValue(
            dv.fileLink(canvasPath, false, "🗂️ " + ch),
            canvasSpan,
            null,
            currentPath,
            true
        );
    });

    // 汇总
    const oldSummary = container.querySelector(".summary-row");
    if (oldSummary) oldSummary.remove();
    const summary = container.createEl("div", { cls: "summary-row", attr: { style: "margin-top: 12px; color: var(--text-muted); font-size: 0.85em;" } });
    summary.setText(`共 ${chapters.length} 个章节白板`);
}

render();
```
