# 🌍 英语
```dataviewjs
// --- 1. 配置与状态管理 ---
const rootPaths = {
    "英语": "考研/英语"
};

if (!state) {
    var state = { subject: "英语", chapter: "" };
}

const mainContainer = dv.container.createEl("div", { id: "math-dashboard-v7" });

// --- 2. 渲染函数 ---
async function render() {
    mainContainer.empty(); 

    // 【删除】原“选择科目”部分（标题、按钮行）

    // 获取数据
    const allPages = dv.pages(`"${rootPaths[state.subject]}"`).filter(p => p.章);
    const chapters = [...new Set(allPages.map(p => p.章).array())].sort((a,b) => {
        const getNum = s => parseInt(s.match(/\d+/) || 0);
        return getNum(a) - getNum(b);
    });

    if (!state.chapter || !chapters.includes(state.chapter)) {
        state.chapter = chapters[0] || "";
    }

    // --- 第二部分：章节选择（原样保留）---
    if (chapters.length > 0) {
        mainContainer.createEl("h2", { text: "📖 选择章节", cls: "math-section-title" });
        const chipRow = mainContainer.createEl("div", { cls: "chip-row" });

        chapters.forEach(ch => {
            const isActive = state.chapter === ch;
            const chip = chipRow.createEl("span", { 
                text: ch, 
                cls: `math-chip ${isActive ? "is-active" : ""}`
            });
            chip.onclick = () => { state.chapter = ch; render(); };
        });
    }

    // 分隔线
    mainContainer.createEl("hr", { attr: { style: "margin: 20px 0; opacity: 0.1;" }});

    // --- 第三部分：内容展示（原样保留）---
    const chapterPages = allPages.filter(p => p.章 === state.chapter);
    let sectionMap = {};

    chapterPages.forEach(p => {
        const secName = p.节;
        if (!secName) return;
        if (!sectionMap[secName]) sectionMap[secName] = { index: 999, link: secName, kps: [] };

        if (p.file.etags.some(tag => tag.includes("节"))) {
            sectionMap[secName].link = p.file.link;
            sectionMap[secName].index = Number(p.index) || sectionMap[secName].index;
        } else if (p.file.etags.includes("#知识点")) {
            if (!sectionMap[secName].kps.some(k => k.link.path === p.file.link.path)) {
                sectionMap[secName].kps.push({ link: p.file.link, index: Number(p.index) || 999 });
            }
            if (sectionMap[secName].index === 999 && p.index) sectionMap[secName].index = Number(p.index);
        }
    });

    const sortedSections = Object.values(sectionMap)
        .filter(s => s.kps.length > 0 || typeof s.link !== "string")
        .sort((a, b) => a.index - b.index);

    // 构建表格
    const table = mainContainer.createEl("table", { cls: "math-table" });
    const thead = table.createEl("thead").createEl("tr");
    thead.createEl("th", { text: "节名称", attr: { style: "width: 25%;" }});
    thead.createEl("th", { text: "核心知识点" });

    const tbody = table.createEl("tbody");
    const currentPath = dv.current().file.path;

    sortedSections.forEach(s => {
        const row = tbody.createEl("tr");
        const tdSec = row.createEl("td", { attr: { style: "text-align: center; font-weight: bold; color: var(--text-accent);" }});
        
        if (typeof s.link === "string") tdSec.setText(s.link);
        else dv.api.renderValue(s.link, tdSec, null, currentPath, true);

        const tdKp = row.createEl("td");
        const kpContainer = tdKp.createEl("div", { attr: { style: "display: flex; flex-wrap: wrap; gap: 8px;" }});
        
        if (s.kps.length === 0) {
            kpContainer.createEl("span", { text: "暂无关联知识点", attr: { style: "color: var(--text-faint); font-size: 0.8em;" }});
        } else {
            s.kps.sort((a, b) => a.index - b.index).forEach(kp => {
                const kpSpan = kpContainer.createEl("span", { cls: "kp-tag" });
                dv.api.renderValue(kp.link, kpSpan, null, currentPath, true);
            });
        }
    });
}

render();
```
