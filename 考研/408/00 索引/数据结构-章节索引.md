```dataviewjs
// 1. 获取并初步过滤
let pages = dv.pages('"考研/408/01 数据结构"')
    .filter(p => p.章 && p.节)
    .array(); // 建议转为数组处理

// 2. 聚合数据
let dataMap = {};
for (let p of pages) {
    if (!dataMap[p.章]) dataMap[p.章] = {};
    if (!dataMap[p.章][p.节]) {
        dataMap[p.章][p.节] = { index: 9999, sectionLink: p.节, kps: [] };
    }

    // 统一转为数字处理，防止 10 < 2 的情况
    let currentIndex = Number(p.index) || 9999;

    if (p.file.etags.includes("#节")) {
        dataMap[p.章][p.节].sectionLink = dv.fileLink(p.file.path, false, p.节);
        dataMap[p.章][p.节].index = currentIndex; 
    } else {
        dataMap[p.章][p.节].kps.push(dv.fileLink(p.file.path, false, p.名称 || p.file.name));
    }
}

// 3. 构建表格行
let rows = [];
// 如果章名也是 "第一章", "第二章"，建议也给章笔记加个 index 排序，
// 否则这里依然会按字母排（第一章、第七章、第三章...）
let chapterKeys = Object.keys(dataMap).sort(); 

for (let chapterName of chapterKeys) {
    let firstInChapter = true;
    
    // 关键修正点：确保按数字大小排
    let sortedSections = Object.keys(dataMap[chapterName])
        .map(secName => ({ name: secName, ...dataMap[chapterName][secName] }))
        .sort((a, b) => a.index - b.index);

    for (let sec of sortedSections) {
        rows.push([
            firstInChapter ? `**${chapterName}**` : "",
            `_ ${sec.sectionLink} _`,
            sec.kps.join("<br>") || "（暂无知识点）"
        ]);
        firstInChapter = false;
    }
}

dv.table(["章", "节", "名称"], rows);
```
