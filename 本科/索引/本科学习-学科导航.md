// --- 1. 配置与状态管理 ---
const rootPaths = {
	"大二下": "本科/大二下",
	"大三上": "本科/大三上",
    "大三下": "本科/大三下"
};

if (!window.state408) {
    window.state408 = { subject: "数据结构", chapter: "" };
}
const state = window.state408;

const mainContainer = dv.container.createEl("div", { id: "math-dashboard-v7" });

// --- 辅助函数：智能数字解析 ---
function parseChapterNum(title) {
    if (!title) return 999;
    const titleStr = String(title);
    const digitMatch = titleStr.match(/^(\d+)/);
    if (digitMatch) return parseInt(digitMatch[1]);
    
    const match = titleStr.match(/第(.+?)[章节]/);
    if (match) {
        const numStr = match[1];
        if (!isNaN(numStr)) return parseInt(numStr);
        const chineseMap = { "零":0, "一":1, "二":2, "三":3, "四":4, "五":5, "六":6, "七":7, "八":8, "九":9, "十":10 };
        if (numStr.length === 1) return chineseMap[numStr] || 999;
        if (numStr.startsWith("十")) return 10 + (chineseMap[numStr[1]] || 0);
        if (numStr.endsWith("十")) return (chineseMap[numStr[0]] || 0) * 10;
        return 999;
    }
    const chineseMap = { "零":0, "一":1, "二":2, "三":3, "四":4, "五":5, "六":6, "七":7, "八":8, "九":9, "十":10 };
    if (chineseMap[titleStr] !== undefined) return chineseMap[titleStr];
    return 999;
}

// --- 2. 渲染函数 ---
async function render() {
    mainContainer.empty(); 

    // --- 第一部分：科目选择 ---
    mainContainer.createEl("h2", { text: "📚 选择科目", cls: "math-section-title" });
    const subRow = mainContainer.createEl("div", { 
        cls: "sub-row",
        attr: { style: "grid-template-columns: repeat(4, 1fr) !important;" } 
    });

    Object.keys(rootPaths).forEach(sub => {
        const isActive = state.subject === sub;
        const btn = subRow.createEl("div", { 
            text: sub, 
            cls: `math-btn ${isActive ? "is-active" : ""}`
        });
        btn.onclick = () => { state.subject = sub; state.chapter = ""; render(); };
    });

    const allPages = dv.pages(`"${rootPaths[state.subject]}"`).filter(p => p.章);
    
    const chapters = [...new Set(allPages.map(p => p.章).array())].sort((a, b) => {
        return parseChapterNum(a) - parseChapterNum(b);
    });

    if (!state.chapter || !chapters.includes(state.chapter)) {
        state.chapter = chapters[0] || "";
    }

    // --- 第二部分：章节选择 ---
    if (chapters.length > 0) {
        mainContainer.createEl("h2", { text: "📖 选择章节", cls: "math-section-title" });
        const chipRow = mainContainer.createEl("div", { cls: "chip-row" });

        chapters.forEach(ch => {
            const isActive = state.chapter === ch;
            const chip = chipRow.createEl("span", { 
                text: ch, 
                cls: `math-chip ${isActive ? "is-active" : ""}`
            });
            chip.onclick = () => { state.chapter = ch; render(); };
        });
    }

    mainContainer.createEl("hr", { attr: { style: "margin: 20px 0; opacity: 0.1;" }});

    // --- 第三部分：内容展示 ---
    const chapterPages = allPages.filter(p => p.章 === state.chapter);
    let sectionMap = {};

    chapterPages.forEach(p => {
        // --- 核心逻辑：节名称归一化 ---
        let rawSec = p.节 || p.名称 || p.file.name;
        if (rawSec && typeof rawSec === 'object' && rawSec.path) {
            rawSec = rawSec.path.split('/').pop().replace('.md', '');
        }
        const rawSecStr = String(Array.isArray(rawSec) ? rawSec[0] : rawSec).trim();

        // 提取“第X节”作为合并Key（支持中文及数字节号）
        const secMatch = rawSecStr.match(/^(第.+?节)/); 
        const secKey = secMatch ? secMatch[1] : rawSecStr;
        
        if (!sectionMap[secKey]) {
            sectionMap[secKey] = { 
                displayName: rawSecStr, 
                link: null, 
                kps: [], 
                minIndex: 999
            };
        }

        const pIndex = (p.index != null) ? parseInt(p.index) : 999;
        const rawTags = (p.tags || []).concat(p.file.tags.array() || []);
        const isMainSection = rawTags.some(t => String(t).includes("章节"));
        
        if (isMainSection) {
            sectionMap[secKey].link = p.file.link;
            sectionMap[secKey].displayName = rawSecStr;
            sectionMap[secKey].minIndex = pIndex; 
        } else {
            // 如果不是主章节，只在 minIndex 是默认值时更新，防止覆盖主章节的 index
            if (sectionMap[secKey].minIndex === 999) {
                sectionMap[secKey].minIndex = pIndex;
            }
        }

        if (p.file.etags.some(t => t.includes("知识点"))) {
            if (!sectionMap[secKey].kps.some(k => k.link.path === p.file.link.path)) {
                sectionMap[secKey].kps.push({ link: p.file.link, index: pIndex });
            }
        }
    });

    const sortedSections = Object.values(sectionMap)
        .map(s => {
            let finalIndex = s.minIndex;
            if (finalIndex === 999) finalIndex = parseChapterNum(s.displayName);
            return { ...s, index: finalIndex };
        })
        .filter(s => s.kps.length > 0 || s.link)
        .sort((a, b) => a.index - b.index);

    const table = mainContainer.createEl("table", { cls: "math-table" });
    const thead = table.createEl("thead").createEl("tr");
    thead.createEl("th", { text: "节名称", attr: { style: "width: 25%;" }});
    thead.createEl("th", { text: "核心知识点" });

    const tbody = table.createEl("tbody");
    const currentPath = dv.current().file.path;

    sortedSections.forEach(s => {
        const row = tbody.createEl("tr");
        const tdSec = row.createEl("td", { attr: { style: "text-align: center; font-weight: bold;" }});
        
        if (s.link) {
            dv.api.renderValue(s.link, tdSec, null, currentPath, true);
        } else {
            tdSec.setText(s.displayName);
        }

        const tdKp = row.createEl("td");
        const kpContainer = tdKp.createEl("div", { attr: { style: "display: flex; flex-wrap: wrap; gap: 8px;" }});
        
        if (s.kps.length === 0) {
            kpContainer.createEl("span", { text: "暂无关联知识点", attr: { style: "color: var(--text-faint); font-size: 0.8em;" }});
        } else {
            s.kps.sort((a, b) => {
                const idxA = (a.index === 999) ? parseChapterNum(a.link.path) : a.index;
                const idxB = (b.index === 999) ? parseChapterNum(b.link.path) : b.index;
                return idxA - idxB;
            }).forEach(kp => {
                const kpSpan = kpContainer.createEl("span", { cls: "kp-tag" });
                dv.api.renderValue(kp.link, kpSpan, null, currentPath, true);
            });
        }
    });
}

render();
