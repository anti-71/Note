```dataviewjs
// 获取所有符合条件的笔记
let pages = dv.pages('"本科/索引" and #学科');

// 1. 按照 index 升序排列（确保原始数据是有序的数字排序）
pages = pages.sort(p => p.index, 'asc');

// 2. 按照“学习阶段”进行分组
let groups = pages.groupBy(p => p["学习阶段"]);

// 3. 对分组后的“大行”进行二次排序，确保学期顺序正确
groups.sort(g => g.rows[0].index, 'asc');

// 4. 渲染表格
dv.table(
    ["阶段", "学科"], 
    groups.map(g => [
        g.key, 
        g.rows.map(p => dv.fileLink(p.file.path, false, p.名称 || p.file.name)).join("<br>")
    ])
);
```
