##### 需求

100 名员工抢 200 个红包，小红包 [1-30] 元占 80%，大红包 [31-100] 元占 20%

##### 实现思路

- 100 个员工 = 100 个线程，竞争 200 个红包
- 用 `synchronized` 同步保证线程安全

##### 代码

```java
public class ThreadTest {
    public static void main(String[] args) {
        List<Integer> redPacket = getRedPacket();
        for (int i = 1; i <= 100; i++) {
            new PeopleGetRedPacket(redPacket, "人" + i).start();
        }
    }

    public static List<Integer> getRedPacket() {
        Random r = new Random();
        List<Integer> redPacket = new ArrayList<>();
        for (int i = 0; i < 160; i++)
            redPacket.add(r.nextInt(30) + 1);   // 小红包
        for (int i = 0; i < 40; i++)
            redPacket.add(r.nextInt(70) + 31);  // 大红包
        return redPacket;
    }
}

class PeopleGetRedPacket extends Thread {
    private List<Integer> redPacket;

    public PeopleGetRedPacket(List<Integer> redPacket, String name) {
        super(name);
        this.redPacket = redPacket;
    }

    @Override
    public void run() {
        while (true) {
            synchronized (redPacket) {
                if (redPacket.size() == 0) break;
                int index = (int)(Math.random() * redPacket.size());
                Integer money = redPacket.remove(index);
                System.out.println(getName() + "抢到了" + money + "元");
            }
            try { Thread.sleep(10); } catch (Exception e) { }
        }
    }
}
```
