df = pd.read_csv('retail_data_2.csv', encoding='latin1', dtype={'Customer': str})

R_value = df3.groupby('CustomerID')['Date'].max()

##### 客户分类

```python
def trans_value(x):
    if x == '高高高': return '重要价值客户'
    elif x == '高低高': return '重要发展客户'
    elif x == '低高高': return '重要保持客户'
    elif x == '低低高': return '重要挽留客户'
    elif x == '高高低': return '一般价值客户'
    elif x == '高低低': return '一般发展客户'
    elif x == '低高低': return '一般保持客户'
    else: return '一般挽留客户'

rfm['客户等级'] = rfm['value'].apply(trans_value)
```
