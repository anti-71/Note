```dataviewjs
// 1. 设置目标文件夹
const folderPath = '"学习安排"';

// 2. 获取任务数据
let tasks = dv.pages(folderPath).file.tasks
.where(t => t.text.includes("-做题:"));

// 【已修复】标准且安全的正则表达式，完美匹配：学科-做题: 书名: [章节] ( 正确 / 总数 )
const regex = /^(.+?)-做题:\s*(.+?):\s*\[(.+?)\]\s*\(\s*(\d+\.?\d*)\s*\/\s*(\d+\.?\d*)\s*\)/;

// 存储聚合数据的对象
let aggregatedData = {};

// 3. 定义子科目排序权重 (数值越小越靠前)
const subSubjectWeight = {
// 数学类
"高数": 1, "线代": 2, "概率论": 3, "套卷": 4,
// 408类
"数据结构": 1, "计算机组成原理": 2, "操作系统": 3, "计算机网络": 4
};

for (let task of tasks) {
const match = task.text.match(regex);

if (match) {
    let subject = match[1].trim();
    let bookName = match[2].trim();
    
    // 排除 数学 的 错题集
    if (subject.includes("数学") && bookName === "错题集") {
        continue; 
    }

    let originalChapter = match[3].trim();
    let correct = parseFloat(match[4]);
    let total = parseFloat(match[5]);
    
    let finalChapter = originalChapter;

    // 章节处理规则
    if (subject.includes("408")) {
        const match408 = originalChapter.match(/^(\d+)/);
        if (match408) finalChapter = `第 ${match408[1]} 章`;
    } else if (subject.includes("英语")) {
        const matchEng = originalChapter.match(/^(\d{4})\s*年/);
        if (matchEng) finalChapter = `${matchEng[1]} 年`;
    }

    const key = `${subject}|${bookName}|${finalChapter}`;
    
    if (!aggregatedData[key]) {
        // 提取子科目名称 (例如从 "数学-高数" 提取 "高数")
        let subMatch = subject.split("-");
        let subName = subMatch.length > 1 ? subMatch[1] : subject;
        
        aggregatedData[key] = {
            subject,
            subName, 
            bookName,
            chapter: finalChapter,
            totalCorrect: 0,
            totalCount: 0,
            weight: subSubjectWeight[subName] || 99 // 未定义的排最后
        };
    }
    
    aggregatedData[key].totalCorrect += correct;
    aggregatedData[key].totalCount += total;
}
}

const allData = Object.values(aggregatedData);

// 4. 定义通用的渲染函数
const renderTable = (title, filterKeyword, excludeKeyword = null) => {
let filtered = allData.filter(item => item.subject.includes(filterKeyword));

if (excludeKeyword) {
    filtered = filtered.filter(item => !item.subject.includes(excludeKeyword));
}

if (filtered.length > 0) {
    filtered.sort((a, b) => {
        if (a.weight !== b.weight) return a.weight - b.weight;
        if (a.bookName !== b.bookName) return a.bookName.localeCompare(b.bookName);
        return a.chapter.localeCompare(b.chapter, undefined, {numeric: true});
    });
    
    dv.header(3, title);
    dv.table(
        ["学科", "书名", "章节", "正确率"], 
        filtered.map(item => {
            let rate = "0.00%";
            if (item.totalCount > 0) {
                rate = ((item.totalCorrect / item.totalCount) * 100).toFixed(2) + "%";
            }
            return [item.subject, item.bookName, item.chapter, `**${rate}**` ];
        })
    );
}
};

// 5. 执行分类渲染
renderTable("数学-习题册 正确率统计", "数学", "套卷");
renderTable("数学-套卷 正确率统计", "数学-套卷");
renderTable("408 正确率统计", "408");
renderTable("英语 正确率统计", "英语");
```
